% matching the app's UI only holds if the type matches.
\usepackage{fontspec}
% __SRS_FONTPATH__ is replaced by compile() in main.js with the absolute path
% to resources/fonts/ (trailing slash, forward slashes even on Windows — XeTeX
% accepts them) before every compile; on a dev checkout where main.js never
% substitutes it, the literal token is not a valid path and fontspec falls
% through to its normal by-name OS lookup (fontconfig on Linux/macOS,
% DirectWrite on Windows) with only a harmless warning in the log. Loading the
% same three weights from a bundled file, by exact filename, means a packaged
% app never depends on DejaVu being installed on the machine it runs on.
\setmainfont{DejaVuSans}[
  Path = __SRS_FONTPATH__,
  Extension = .ttf,
  UprightFont = *,
  BoldFont = *-Bold,
  ItalicFont = *-Oblique,
  BoldItalicFont = *-BoldOblique,
  Scale = 0.96,
]

% Warnings are a table too, built from \parbox rows like the interface table
% below and for the identical reason: \uiwarning opens one warning record and
% the \uiwarnrich call(s) that may follow attach conditions to it, so the row's
% content is only complete once the NEXT \uiwarning (or \srswarnend) confirms
% no more \uiwarnrich is coming for it — a real tabular's row cannot be held
% open across those separate macro calls. So each record is harvested into
% \warnval@* and flushed as one row at that point.
\newlength{\srswarndesc}
\newlength{\srswarngap}
\newif\ifsrswarnpending
\makeatletter

  \gdef\warnval@id{}\gdef\warnval@on{}\gdef\warnval@off{}%
  \gdef\warnval@enter{}\gdef\warnval@exit{}%
}
\newcommand{\srswarnsetup}{%
  \setlength{\srswarngap}{5pt}%
  % Fixed columns: 2.4cm (badge) + 1.6cm + 1.6cm = 5.6cm, plus 4 inter-column
  % gaps; whatever is left splits evenly between the two condition columns.
  \setlength{\srswarndesc}{\dimexpr(\linewidth-5.6cm-4\srswarngap)/2\relax}%
}
\newcommand{\srswarncells}[5]{%
  \parbox[t]{2.4cm}{\raggedright\strut #1}\hspace{\srswarngap}%
  \parbox[t]{1.6cm}{\raggedright\strut #2}\hspace{\srswarngap}%
  \parbox[t]{1.6cm}{\raggedright\strut #3}\hspace{\srswarngap}%
  \parbox[t]{\srswarndesc}{\raggedright\strut #4}\hspace{\srswarngap}%
  \parbox[t]{\srswarndesc}{\raggedright\strut #5}%
  \par
}
\newcommand{\srswarnflush}{%
  \ifsrswarnpending

  \fi
}

% Choose the branch BEFORE anything starts scanning. Leaving \begin{tabular}
% between \else and \fi lets the tabular swallow the \fi into its body and
% TeX reports "Incomplete \iftrue" far from the real cause.
% \ifinitemprops tells \itemrich (below) whether the tabular from
% \srsprops@open is currently the open one, so a compact rich field (Master/
% Slave) knows to print its own row instead of a standalone rich block.
\newif\ifinitemprops
\makeatletter
\newcommand{\srsprops@open}{%
  \initempropstrue
  \par\smallskip
  % \strut in BOTH column preambles is load-bearing. Without it the two
  % p-columns get first lines of different heights (sans has taller ascenders
  % than serif at the same size), and since a p-column's baseline is its first
  % line's baseline, label and value end up visibly out of line. tabularx would
  % also fix it but cannot be used here: it scans for \end{tabularx} in the
  % token stream, which fails when \begin/\end are split across an environment
  % definition's two halves.
  \begingroup\small\setlength{\parskip}{0pt}\renewcommand{\arraystretch}{1.2}%
  \begin{tabular}{@{}>{\strut\sffamily\bfseries\color{srsmuted}\raggedright\arraybackslash}p{4.4cm}@{\hspace{1em}}>{\strut\raggedright\arraybackslash}p{\dimexpr\linewidth-5.4cm\relax}@{}}
}

  \end{tabular}%
  \endgroup\par\smallskip
  \initempropsfalse
}
